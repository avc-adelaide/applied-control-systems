	Define a PID controller $C$ with gains $K_p=1$, $\TTi=2.9$, $\TTd=0.9$, $N=100$. \\[10pt]

\begin{frame}[fragile]
\frametitle{Sensitivity functions in Matlab}
Calculate the Gang of Four transfer functions in Matlab
\begin{itemize}
	\item The sensitivity function is 
\[S=\frac{1}{1+PC}\]
\item The load (or input) sensitivity function is 
\[PS=\frac{P}{1+PC}\]
\item The complementary sensitivity function is 
\[T=\frac{PC}{1+PC}\]
\item The noise (or output) sensitivity function is
\[CS = \frac{C}{1+PC}\]
\end{itemize}
\end{frame}
